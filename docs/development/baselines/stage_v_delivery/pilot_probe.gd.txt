extends SceneTree
func _initialize() -> void:
 var packed:PackedScene=load("res://assets/environment/exports/structures/industrial_wall_4m.glb")
 for j in range(2):
  var wall := packed.instantiate()
  root.add_child(wall)
  for node in wall.find_children("*","MeshInstance3D",true,false):
   print("PILOT AABB ",node.get_aabb()," BASIS ",node.transform)
   var geometry := ""
   for i in range(node.mesh.get_surface_count()):geometry += str(node.mesh.surface_get_arrays(i)[Mesh.ARRAY_VERTEX])
   print("PILOT GEOMETRY ",geometry.sha256_text())
   for i in range(node.mesh.get_surface_count()):print("PILOT MAT ",node.mesh.surface_get_material(i).resource_path)
  wall.queue_free()
 quit()
