extends "res://tests/expanded_world_test.gd"
func run() -> void:
 test_directory="res://.godot/w_decision_followup/gui_" + Crypto.new().generate_random_bytes(8).hex_encode()
 DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(test_directory))
 var arena := create_u("gui")
 await frames(15)
 await preview("briefing")
 arena.objective_view.deploy_button.pressed.emit()
 await frames(2)
 check(arena.deployment_started,"Graphical Deploy button signal starts expanded mission")
 freeze(arena)
 await frames(12)
 arena.process_entries()
 freeze(arena)
 park_world(arena)
 var victim:Raider=arena.raiders[1]
 victim.position=Vector3(0,.1,20)
 arena.robot.position=Vector3(0,.1,17)
 arena.ledger.collect(1,&"Salvage",3)
 await frames(3)
 var weapon:ShotExecutor=arena.robot.get_node("Weapon")
 for i in range(8):
  if victim.health.current<=0:break
  weapon.cooldown=0
  weapon.try_fire_at(victim.position+Vector3.UP)
 check(victim.health.current==0,"Player weapon physically defeats a loaded Raider in W")
 arena.robot.position=Vector3(0,.1,20)
 await frames(3)
 check(arena.try_recover(),"Player physically recovers W kill cargo")
 await preview("active")
 await finish_recall(arena)
 check(arena.mission.player_extracted and arena.ledger.banked_materials==3,"Graphical W mission completes objectives and banks finite cargo")
 await preview("success")
 print("Follow-up GUI artifacts: ",test_directory)
 arena.queue_free()
 await frames(30)
 quit(1 if failures else 0)
