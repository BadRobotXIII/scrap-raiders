extends "res://tests/expanded_world_test.gd"

func capture(label: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://.godot/stage_z_audit/" + label + ".png")

func run() -> void:
	test_directory = "res://.godot/stage_z_audit/gui_" + Crypto.new().generate_random_bytes(8).hex_encode()
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(test_directory))
	var arena := create_u("expanded_gui")
	await frames(12)
	await capture("default_briefing")
	arena.objective_view.deploy_button.pressed.emit()
	await frames(2)
	freeze(arena)
	await frames(12)
	arena.process_entries()
	freeze(arena)
	park_world(arena)
	var victim: Raider = arena.raiders[1]
	victim.position = Vector3(-9, 0.1, 24)
	arena.robot.position = Vector3(-9, 0.1, 18)
	arena.ledger.collect(1, &"Salvage", 3)
	await frames(45)
	var weapon: ShotExecutor = arena.robot.get_node("Weapon")
	for index in range(4):
		weapon.cooldown = 0
		weapon.try_fire_at(victim.position + Vector3.UP)
		await capture("default_hit_%d" % index)
		await frames(10)
	check(not is_instance_valid(victim) or victim.health.current == 0, "Graphical physical fire defeats a loaded Raider")
	arena.robot.position = Vector3(-9, 0.1, 24)
	await frames(3)
	check(arena.try_recover(), "Graphical finite cargo recovery succeeds")
	await capture("default_recovery")
	
	await finish_recall(arena)
	check(arena.mission.player_extracted and arena.ledger.banked_materials == 3, "Graphical Recall banks objective cargo")
	await frames(3)
	await capture("default_result")
	arena.restart_mission()
	await frames(12)
	arena = current_scene
	check(not arena.deployment_started and get_nodes_in_group("presentation_bursts").is_empty(), "Graphical restart clears effects and restores briefing")
	arena.start_deployment()
	await frames(30)
	arena.robot.get_node("Health").apply_damage(1000)
	check(arena.post_death.enter(), "Graphical post-death follow remains available")
	await frames(20)
	arena.post_death.cycle(1)
	check(arena.post_death.eligible(arena.post_death.selected), "Graphical follow cycles to a surviving autonomous actor")
	await capture("default_post_death")
	arena.queue_free()
	await frames(30)
	print("Stage Y graphical failures: ", failures)
	quit(1 if failures else 0)
