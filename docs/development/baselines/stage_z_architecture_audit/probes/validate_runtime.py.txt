from pathlib import Path
import subprocess,json
out=Path('.godot/stage_z_audit')
source=Path('.godot/stage_y_refinement/delivery')
for name in ['graphical.gd','replay.gd']:
    s=(source/name).read_text(encoding='utf-8').replace('res://.godot/stage_y_refinement/delivery/','res://.godot/stage_z_audit/')
    (out/name).write_text(s,encoding='utf-8')
rows=[]
for label,args,marker in [
    ('baseline_seeded',['--headless','--verbose','--path','.','--fixed-fps','60','--script','res://.godot/stage_z_audit/replay.gd','--quit-after','20000'],'Y REPLAY COMPLETE 0'),
    ('baseline_graphical',['--path','.','--script','res://.godot/stage_z_audit/graphical.gd','--quit-after','20000'],'Stage Y graphical failures: 0'),
]:
    with (out/(label+'.log.txt')).open('w',encoding='utf-8') as f:
        r=subprocess.run([r'C:\Users\kzulf\bin\godot.cmd',*args],stdout=f,stderr=subprocess.STDOUT,timeout=180)
    log=(out/(label+'.log.txt')).read_text(encoding='utf-8',errors='replace')
    row={'case':label,'exit':r.returncode,'completed':marker in log,'issues':[line for line in log.splitlines() if 'ERROR:' in line or 'WARNING:' in line]}
    rows.append(row)
    (out/'runtime_validation.json').write_text(json.dumps(rows,indent=2),encoding='utf-8')
    print(row,flush=True)
    assert r.returncode==0 and row['completed']
