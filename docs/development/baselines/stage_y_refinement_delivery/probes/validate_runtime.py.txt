from pathlib import Path
import subprocess, json, shutil
out=Path('.godot/stage_y_refinement/delivery')
rows=[]
cases=[]
for view in ['bay','overview']:
    label='yr_final_'+view
    cases.append((label,['--path','.','--script','res://'+(out/'performance.gd').as_posix(),'--quit-after','20000','--',label,'res://scenes/levels/expanded_salvage_complex.tscn'],'PERFORMANCE '))
cases += [
    ('inventory',['--path','.','--script','res://'+(out/'inventory.gd').as_posix(),'--quit-after','20000'],'Stage Y inventory failures: 0'),
    ('graphical',['--path','.','--script','res://'+(out/'graphical.gd').as_posix(),'--quit-after','20000'],'Stage Y graphical failures: 0'),
    ('peak',['--path','.','--script','res://.godot/stage_x/peak.gd','--quit-after','20000','--','yr_final_peak'],'PEAK'),
    ('native_population',['--headless','--verbose','--path','.','--script','res://tests/world_population_test.gd','--quit-after','20000'],'World population failures: 0'),
    ('bounded_expanded',['--headless','--path','.','res://scenes/levels/expanded_salvage_complex.tscn','--quit-after','300'],None),
    ('bounded_default',['--headless','--path','.','--quit-after','300'],None),
]
for label,args,marker in cases:
    with (out/(label+'.log.txt')).open('w') as f:
        r=subprocess.run([r'C:\Users\kzulf\bin\godot.cmd',*args],stdout=f,stderr=subprocess.STDOUT,timeout=180)
    log=(out/(label+'.log.txt')).read_text(errors='replace')
    row={'case':label,'exit':r.returncode,'completed':marker is None or marker in log,'issues':[line for line in log.splitlines() if 'ERROR:' in line or 'WARNING:' in line]}
    rows.append(row)
    (out/'runtime_validation.json').write_text(json.dumps(rows,indent=2))
    print(row,flush=True)
    if label.startswith('yr_final_'):
        shutil.copy2(Path('.godot/stage_w/performance')/('performance_'+label+'.json'),out/(label+'.json'))
    if r.returncode:raise SystemExit(1)
