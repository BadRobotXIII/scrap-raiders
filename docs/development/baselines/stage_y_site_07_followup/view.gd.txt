func capture(arena: Node3D, label: String) -> void:
 for view in ["side","end"]:
  arena.spectator.camera.global_position=Vector3(2,4,28) if view=="side" else Vector3(-2,3,32)
  arena.spectator.camera.look_at(Vector3(-5,1.7,24))
  await create_timer(0.1).timeout
  await RenderingServer.frame_post_draw
  root.get_texture().get_image().save_png("res://.godot/stage_y_site_07/"+label+"_"+view+".png")
