extends SceneTree
func _initialize() -> void: call_deferred("run")
func run() -> void:
 var scene := load("res://scenes/characters/actor_inspection.tscn").instantiate() as Node3D
 root.add_child(scene)
 current_scene = scene
 for index in [-1,0,1,2,3,4,5]:
  scene.select_view(index)
  await create_timer(.25).timeout
  await RenderingServer.frame_post_draw
  root.get_texture().get_image().save_png("res://.godot/stage_w/actor_view_%d.png" % index)
 scene.queue_free()
 await process_frame
 quit()
