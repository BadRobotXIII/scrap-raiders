extends "res://tests/inventory_world_test.gd"

func capture(label: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://.godot/stage_z_build/" + label + ".png")

func key(code: Key) -> void:
	for pressed in [true, false]:
		var event := InputEventKey.new()
		event.physical_keycode = code
		event.keycode = code
		event.pressed = pressed
		root.push_input(event)
		await frames(1)

func click(button: Control) -> void:
	var point := button.get_global_rect().get_center()
	for pressed in [true, false]:
		var event := InputEventMouseButton.new()
		event.button_index = MOUSE_BUTTON_LEFT
		event.position = point
		event.global_position = point
		event.pressed = pressed
		root.push_input(event)
		await frames(1)

func run() -> void:
	root.size = Vector2i(1600, 900)
	var arena := create_arena("graphical", true)
	await frames(15)
	await capture("briefing")
	await key(KEY_I)
	check(arena.interaction_mode.view.visible and not arena.robot.input_adapter.gameplay_enabled, "Graphical I opens preparation without device recapture")
	await capture("prepared")
	await key(KEY_ESCAPE)
	check(not arena.interaction_mode.view.visible and not arena.deployment_started, "Graphical Escape closes inventory before handling gameplay capture")
	await click(arena.objective_view.deploy_button)
	await frames(25)
	check(arena.deployment_started and not arena.mission.is_finished(), "Real GUI Deploy commits inventory and starts the expanded world")
	freeze(arena)
	var source: WorldLoot = arena.drops[-1]
	arena.robot.position = source.position + Vector3(0, 0.1, 1)
	await frames(5)
	arena.interaction_mode.open_inventory(source)
	arena.inventory_interaction.opened = source
	await capture("source")
	arena.interaction_mode.view.perform("Take All")
	await capture("looted")
	arena.interaction_mode.close_inventory()
	arena.robot.position = arena.get_node("RecallDock").position + Vector3.UP * 0.1
	arena.robot.velocity = Vector3.ZERO
	await frames(12)
	arena.robot.set_physics_process(false)
	arena.robot.input_adapter.set_physics_process(false)
	arena.robot.input_adapter.recover_held = true
	arena.recall.step(0.1)
	for index in 60:
		if arena.mission.is_finished(): break
		arena.recall.step(0.1)
	check(arena.mission.player_extracted, "Graphical source loot and real Recall complete durable settlement")
	await capture("result")
	arena.restart_mission()
	await frames(15)
	arena = current_scene
	arena.start_deployment()
	await frames(30)
	arena.robot.get_node("Health").apply_damage(1000)
	check(arena.post_death.enter(), "Graphical post-death follow remains available")
	await frames(12)
	await capture("post_death")
	arena.queue_free()
	await frames(30)
	print("Stage Z graphical failures: ", failures)
	quit(1 if failures else 0)
