var shot_log: Array = []
var transfer_log: Array = []
var weapons: Dictionary = {}
func fired(a: Vector3, b: Vector3, damaged: bool, id: String) -> void:
 shot_log.append([id,a,b,damaged])
func transfer(kind: StringName, details: Dictionary, source: String) -> void:
 transfer_log.append([source,kind,details.duplicate(true)])
func hook(arena: Node3D) -> void:
 for id in arena.raiders:
  var actor = arena.raiders[id]
  if is_instance_valid(actor) and not weapons.has(id):
   weapons[id]=true
   actor.combat.weapon.fired.connect(fired.bind("R%d" % id))
