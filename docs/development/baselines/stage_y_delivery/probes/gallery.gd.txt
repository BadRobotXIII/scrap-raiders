extends SceneTree
func _initialize() -> void:call_deferred("run")
func run() -> void:
 var scene=load("res://scenes/environment/asset_inspection.tscn").instantiate()
 root.add_child(scene)
 for i in range(20):await process_frame
 await RenderingServer.frame_post_draw
 root.get_texture().get_image().save_png("res://.godot/stage_y/asset_inspection.png")
 scene.queue_free()
 for i in range(5):await process_frame
 print("Y INSPECTION CAPTURE COMPLETE")
 quit()
